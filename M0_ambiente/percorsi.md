# SECONDO ESERCIZIO

biseogna usare percorsi relativi e assoluti, creazione e spostamento in cartelle 
# qua entro nella mia cartella personale
```
cd Users\aysha.mohammed\Documents\
```

# qua entro all'interno della cartella documenti
```
cd Documenti
```

#  una volta dentro la cartella, vado a creare la cartella esercizio-percorsi
```
mkdir esercizio-percorsi
```

# vado a creare al suo interno delle sottocartelle
```
mkdir esercizio-percorsi\dati
```

# creo l'altra sottocartella sempre all'interno della cartella esercizio-percorsi
```
mkdir esercizio-percorsi\risultati
```

# mi sposto all'interno della sottocartella dati che si trova all'interno della cartella esercizio-percorsi
```
cd esercizio-percorsi\dati
```

# arriviamo alla sottocartella risultati
```
cd ..\risultati
```
